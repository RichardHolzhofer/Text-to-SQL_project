> User: Show me the top 5 customers by revenue this year
SELECT
  c.customer_name,
  SUM(o.total_amount) as revenue
FROM customers c
JOIN orders o ON c.id = o.customer_id
WHERE EXTRACT(YEAR FROM o.order_date) = EXTRACT(YEAR FROM CURRENT_DATE)
GROUP BY 1
ORDER BY 2 DESC
LIMIT 5;
